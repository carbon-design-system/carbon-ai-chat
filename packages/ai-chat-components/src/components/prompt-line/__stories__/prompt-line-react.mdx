## Overview

The prompt line is the **input interface component for the chat**. It provides the editing surface where users compose and send messages. It scales from a simple text input up to a rich editor with autocomplete suggestions, context chips, file attachments, and custom extensions.

The prompt line supports:

- **Text input** — text entry with placeholder, disabled, and autofocus states.
- **Autocomplete** — trigger-driven suggestion overlays via `useChatAutocomplete` and the `buildCarbonExtensions` helper.
- **Message actions** — icon action buttons projected into the `message-actions` slot (e.g. attach file, switch mode).
- **File uploads** — a visual file chip list projected into the `file-uploads` slot via `<CDSAIChatFileUploads>`.
- **Context chips** — inline token chips rendered inside the rich editor (e.g. `@mention`, `/command`).
- **Error state** — displays error message `field-messaging` slot and applies error styling.

It composes three React components into a complete input surface:

- `<PromptLineShell>` - layout chrome that defines named slots and controls inline/expanded layout modes.
- `<PromptLine>` — the editing surface; renders a plain `<textarea>` or a Tiptap rich-text editor with additional capabilities (context chips, custom Tiptap nodes, etc.) when `extensions` are provided.
- `<CDSAIChatInputSendControl>` — renders the send button or stop streaming button.

Additional components used with the prompt line:

- `<CDSAIChatAutocomplete>` - suggestion list overlay rendered by the `useChatAutocomplete` hook in the `autocomplete-content` slot.
- `<CDSAIChatFileUploads>` — container of file attachment chips rendered in the `file-uploads` slot.
- `<CDSAIChatErrorMessage>` — error message rendered in the `field-messaging` slot.

### Getting started

Here's a minimal implementation.

```jsx
import { useState, useCallback } from 'react';
import PromptLine from '@carbon/ai-chat-components/es/react/prompt-line.js';
import PromptLineShell from '@carbon/ai-chat-components/es/react/prompt-line-shell.js';
import CDSAIChatInputSendControl from '@carbon/ai-chat-components/es/react/input-send-control.js';

function ChatInput() {
  const [hasValidInput, setHasValidInput] = useState(false);

  const handleChange = useCallback((e) => {
    setHasValidInput(e.detail.rawValue.length > 0);
  }, []);

  const handleSend = useCallback(() => {
    // submit the message
  }, []);

  return (
    <PromptLineShell rounded>
      <PromptLine
        slot="editor"
        placeholder="Ask a question..."
        onChange={handleChange}
        onSendIntent={handleSend}
      />
      <CDSAIChatInputSendControl
        slot="send-control"
        hasValidInput={hasValidInput}
        onSend={handleSend}
      />
    </PromptLineShell>
  );
}
```

### Using with autocomplete

```jsx
import { useState, useCallback, useMemo, useRef } from 'react';
import PromptLine from '@carbon/ai-chat-components/es/react/prompt-line.js';
import PromptLineShell from '@carbon/ai-chat-components/es/react/prompt-line-shell.js';
import CDSAIChatInputSendControl from '@carbon/ai-chat-components/es/react/input-send-control.js';
import { useChatAutocomplete } from '@carbon/ai-chat-components/es/react/hooks/useChatAutocomplete.js';
import { buildCarbonExtensions } from '@carbon/ai-chat-components/es/components/prompt-line/index.js';

const ALL_ITEMS = [
  { id: '1', label: 'Summarize this document' },
  { id: '2', label: 'Translate to Spanish' },
  { id: '3', label: 'Write a follow-up email' },
];

function ChatInput() {
  const promptLineRef = useRef(null);
  const [hasValidInput, setHasValidInput] = useState(false);

  const autocomplete = useMemo(
    () => ({
      items: (query) =>
        ALL_ITEMS.filter((i) =>
          i.label.toLowerCase().includes(query.toLowerCase())
        ),
    }),
    []
  );

  const extensions = useMemo(
    () => buildCarbonExtensions({ autocomplete }),
    [autocomplete]
  );

  const { onTriggerChange, autocompleteContent } = useChatAutocomplete({
    autocomplete,
    promptLineRef,
  });

  const handleChange = useCallback((e) => {
    setHasValidInput(e.detail.rawValue.length > 0);
  }, []);

  return (
    <PromptLineShell rounded expanded>
      <PromptLine
        ref={promptLineRef}
        slot="editor"
        placeholder="Ask a question..."
        rich
        extensions={extensions}
        onChange={handleChange}
        onTriggerChange={onTriggerChange}
      />
      {autocompleteContent}
      <CDSAIChatInputSendControl
        slot="send-control"
        hasValidInput={hasValidInput}
        onSend={() => {
          /* submit */
        }}
      />
    </PromptLineShell>
  );
}
```

#### Filtering across multiple words

The `query` passed to your `items` function is the **full editor text up to the cursor**, including spaces. The overlay stays open as the user types and the query grows — there is no word boundary that resets or closes it.

Filter against the full query string to match suggestions across words:

```jsx
const autocomplete = useMemo(
  () => ({
    items: (query) =>
      ALL_ITEMS.filter((i) =>
        i.label.toLowerCase().includes(query.toLowerCase())
      ),
  }),
  []
);
```

Because `query` may contain leading or trailing spaces (for example, `"translate "` after the user pauses mid-phrase), trim before comparing if you want to treat `"translate"` and `"translate "` as the same filter:

```jsx
items: (query) => {
  const q = query.trim().toLowerCase();
  return q ? ALL_ITEMS.filter((i) => i.label.toLowerCase().includes(q)) : ALL_ITEMS;
},
```

### Using with file uploads

```jsx
import { useState, useCallback, useRef } from 'react';
import PromptLine from '@carbon/ai-chat-components/es/react/prompt-line.js';
import PromptLineShell from '@carbon/ai-chat-components/es/react/prompt-line-shell.js';
import CDSAIChatInputSendControl from '@carbon/ai-chat-components/es/react/input-send-control.js';
import CDSAIChatFileUploads from '@carbon/ai-chat-components/es/react/file-uploads.js';
import { FileStatusValue } from '@carbon/ai-chat-components/es/components/prompt-line/index.js';

function ChatInput() {
  const fileInputRef = useRef(null);
  const [uploads, setUploads] = useState([]);
  const [hasValidInput, setHasValidInput] = useState(false);

  const handleAttachClick = useCallback(() => {
    fileInputRef.current?.click();
  }, []);

  const handleFileSelected = useCallback((e) => {
    const files = Array.from(e.target.files ?? []);
    e.target.value = ''; // reset so re-selecting the same file fires `change`
    const newUploads = files.map((file) => ({
      id: `file-${Date.now()}-${Math.random().toString(36).slice(2)}`,
      file,
      status: FileStatusValue.EDIT,
    }));
    setUploads((prev) => [...prev, ...newUploads]);
  }, []);

  const handleFileRemove = useCallback((e) => {
    setUploads((prev) => prev.filter((u) => u.id !== e.detail.fileId));
  }, []);

  const hasValidInputOrUploads =
    hasValidInput || uploads.some((u) => !u.isError);

  return (
    <PromptLineShell rounded expanded>
      {/* Always mounted so live regions persist after the last file is removed */}
      <CDSAIChatFileUploads
        slot="file-uploads"
        uploads={uploads}
        onFileRemove={handleFileRemove}
      />
      <PromptLine
        slot="editor"
        placeholder="Ask a question..."
        onChange={(e) => setHasValidInput(e.detail.rawValue.length > 0)}
      />
      <div slot="message-actions">
        <input
          ref={fileInputRef}
          type="file"
          multiple
          tabIndex={-1}
          hidden
          onChange={handleFileSelected}
        />
        <cds-icon-button
          size="sm"
          kind="ghost"
          align="top-start"
          onClick={handleAttachClick}>
          {/* attach icon */}
          <span slot="tooltip-content">Attach file</span>
        </cds-icon-button>
      </div>
      <CDSAIChatInputSendControl
        slot="send-control"
        hasValidInput={hasValidInputOrUploads}
        onSend={() => {
          /* submit */
        }}
      />
    </PromptLineShell>
  );
}
```
