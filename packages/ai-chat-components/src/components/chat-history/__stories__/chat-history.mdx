## Overview

The **`<cds-aichat-history-shell>`** component provides a structured layout for displaying and managing chat conversation history. It defines three dedicated slots that organize the chat history interface:

<table style={{ width: '100%' }}>
  <thead>
    <tr>
      <th>Slot Name</th>
      <th>Description</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>
        <code>header</code>
      </td>
      <td>Represents the header section, containing title and actions.</td>
    </tr>
    <tr>
      <td>
        <code>toolbar</code>
      </td>
      <td>
        Represents the toolbar area of the chat history component (includes
        search, new chat button, etc).
      </td>
    </tr>
    <tr>
      <td>
        <code>content</code>
      </td>
      <td>List of the chat history items.</td>
    </tr>
  </tbody>
</table>

Each of these slots can accept **any element or component**, allowing users to freely construct and customize the chat history layout as needed. Alternatively, adopters can use the **provided subcomponents**, which are designed specifically for each section. These subcomponents offer consistent styling and default behaviors that align with the AI Chat design patterns.

### Delete confirmation panel

Render **`<cds-aichat-history-delete-panel>`** when the user chooses **Delete** from a chat item’s overflow menu. Set **`item-id`** to the id of the chat item being removed; **`cds-aichat-history-shell`** uses this (with **`history-delete-confirm`**) to move keyboard focus to the next or previous chat item after your handler deletes the item, and to fire **`history-item-selected`** on that item when the deleted chat item was the active/selected one.

Listen for **`history-delete-cancel`** to dismiss the panel without changes, and **`history-delete-confirm`** to remove the chat from your data store and hide the panel.

```html
<cds-aichat-history-shell>
  <cds-aichat-history-header header-title="Chats"></cds-aichat-history-header>
  <cds-aichat-history-toolbar></cds-aichat-history-toolbar>
  <cds-aichat-history-content>
    <cds-aichat-history-panel aria-label="Chat history">
      <cds-aichat-history-panel-items>
        <!-- Panel items; overflow Delete opens the delete panel -->
      </cds-aichat-history-panel-items>
    </cds-aichat-history-panel>
  </cds-aichat-history-content>
  <cds-aichat-history-delete-panel item-id="chat-to-delete">
    <div slot="title">Confirm Delete</div>
    <div slot="description">This conversation will be permanently deleted.</div>
  </cds-aichat-history-delete-panel>
</cds-aichat-history-shell>
```

### Validating renamed titles

```javascript
// Listen for item selection
document.addEventListener('history-item-selected', (event) => {
  const { itemId, itemName } = event.detail;
  console.log(`Selected chat: ${itemName} (ID: ${itemId})`);
  loadConversation(itemId);
});

// Listen for item actions
document.addEventListener('history-item-menu-action', (event) => {
  const { action, itemId, itemName, element } = event.detail;

  switch (action) {
    case 'Rename':
      element.rename = true;
      break;
    case 'Delete':
      console.log(`Deleting chat: ${itemName}`);
      break;
    case 'Pin':
      console.log(`Pinning chat: ${itemName}`);
      break;
  }
});

// Validate the rename input on each keystroke
document.addEventListener('history-panel-item-input-change', (event) => {
  const { value, itemId } = event.detail;
  const item = document.querySelector(
    `cds-aichat-history-panel-item[id="${CSS.escape(itemId)}"]`
  );
  if (!item) return;

  const invalid = value.length > 75;
  item.setAttribute('rename-invalid', invalid ? '' : null);
  item.setAttribute(
    'rename-invalid-message',
    invalid ? 'Title cannot exceed 75 characters.' : ''
  );
});

// Persist the new name on save
document.addEventListener('history-panel-item-input-save', (event) => {
  const { newName, itemId } = event.detail;
  updateChatName(itemId, newName);
});
```

### Renaming with a standalone input

```javascript
// Validate on each keystroke and reflect state back to the item
document.addEventListener('history-panel-item-input-change', (event) => {

    `cds-aichat-history-panel-item[id="${CSS.escape(itemId)}"]`

  const { newName, itemId } = event.detail;
  updateChatName(itemId, newName);
```

### Confirming deletion

Set `item-id` whenever you show the delete panel so `HistoryShell` can restore focus after deletion.

```html
<cds-aichat-history-shell>
  <cds-aichat-history-delete-panel item-id="chat-to-delete">
    <div slot="title">Delete this chat?</div>
    <div slot="description">This conversation will be permanently deleted.</div>
  </cds-aichat-history-delete-panel>
</cds-aichat-history-shell>
```

```javascript
// Listen for cancel
document.addEventListener('history-delete-cancel', () => {
  console.log('Delete cancelled');
  // Hide the delete panel
  document.querySelector('cds-aichat-history-delete-panel').remove();
});

// Listen for delete confirmation
document.addEventListener('history-delete-confirm', (event) => {
  const { itemId, nextItemId, deletedItemWasSelected } = event.detail ?? {};
  console.log('Delete confirmed', {
    itemId,
    nextItemId,
    deletedItemWasSelected,
  });
  // Remove the chat item from your data store; the shell handles focus if item-id was set
  deleteChatFromStore(itemId);
  document.querySelector('cds-aichat-history-delete-panel').remove();
});
```

### Focus management

When performing actions that reorder or move history items (such as pinning/unpinning), you may need to restore focus to the item after the DOM updates for accessibility reasons. Use the `focusElementAfterRepaint` utility to handle this:

```javascript
import { focusElementAfterRepaint } from '@carbon/ai-chat-components/es/globals/utils/focus-utils.js';
```

**Example with Pin/Unpin:**

```javascript
import { focusElementAfterRepaint } from '@carbon/ai-chat-components/es/globals/utils/focus-utils.js';

// Listen for pin action
document.addEventListener('history-item-menu-action', (event) => {
  const { action, itemId } = event.detail;

  if (action === 'Pin') {
    // Move item from regular to pinned list in your data
    const itemToPin = regularItems.find((chat) => chat.id === itemId);
    pinnedItems.unshift(itemToPin);
    regularItems = regularItems.filter((chat) => chat.id !== itemId);

    focusElementAfterRepaint(
      this.renderRoot,
      `cds-aichat-history-panel-item[id="${CSS.escape(itemId)}"]`
    );
  }
});
```
